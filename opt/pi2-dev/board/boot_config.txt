# Full firmware required: the cut-down start_cd.elf lacks the ISP service the
# camera pipeline drives over VCHIQ (camera never enumerates -- hw-tested on pi0).
start_file=start.elf
fixup_file=fixup.dat

# Kernel location
kernel=zImage

# How much memory in MB to assign to the GPU on Pi models having
# 32: the GPU only needed 64MB for the legacy MMAL camera stack (start_x=1),
# which libcamera replaces. The ARM-side camera drivers instead allocate
# capture buffers from the kernel's 64MB CMA pool.
gpu_mem=32

# Enable SPI
dtparam=spi=on

# Set the bootloader delay to 0 seconds. The default is 1s if not specified
boot_delay=0
disable_splash=1

### Dev image additions ###

# USB relay: expose the USB port as an ethernet gadget when plugged into a
# computer (see docs/dev_images.md). dwc2 is built dual-role, so a keyboard
# on an OTG adapter still works.
dtoverlay=dwc2

# HDMI console output even when the display is plugged in after boot
hdmi_force_hotplug=1

# libcamera: sensor overlay for the standard SeedSigner camera (v1 / OV5647).
# Use dtoverlay=imx219 for a v2 module. The release image instead used to set
# start_x=1 to boot the GPU-firmware camera stack, which would claim the sensor
# and leave libcamera with no cameras to enumerate.
dtoverlay=ov5647
